Base Model Options (Pre-quantization size → Post-quantization):
- Llama-2-7B (13GB) → Llama-2-7B-4bit (3.5GB) → Fine-tuned (500MB LoRA)
- Mistral-7B (13GB) → Mistral-7B-4bit (3.5GB) → Fine-tuned (500MB LoRA)
- Gemma-2B (4.5GB) → Gemma-2B-4bit (1.2GB) → Fine-tuned (200MB LoRA)

Recommended: Gemma-2B for 500MB target with LoRA

2.3 Training Script with Unsloth
--------------------------------
```python
# train_legal_model_unsloth.py
import torch
from unsloth import FastLanguageModel
from trl import SFTTrainer
from transformers import TrainingArguments
from datasets import load_dataset
import json

def setup_model_for_training():
    # Load 4bit quantized model
    model, tokenizer = FastLanguageModel.from_pretrained(
        model_name="unsloth/gemma-2b-bnb-4bit",  # 4bit quantized
        max_seq_length=2048,
        dtype=None,  # Auto-detect
        load_in_4bit=True,
    )

    # Add LoRA adapters
    model = FastLanguageModel.get_peft_model(
        model,
        r=16,           # LoRA rank (controls model size)
        target_modules=["q_proj", "k_proj", "v_proj", "o_proj",
                       "gate_proj", "up_proj", "down_proj"],
        lora_alpha=16,
        lora_dropout=0.1,
        bias="none",
        use_gradient_checkpointing="unsloth",  # Memory efficient
        random_state=3407,
        use_rslora=False,
        loftq_config=None,
    )

    return model, tokenizer

def prepare_legal_dataset():
    # Load your custom legal dataset
    dataset = load_dataset("json", data_files="legal_training_dataset.jsonl")

    def format_prompts(examples):
        texts = []
        for text in examples["text"]:
            # Add EOS token for proper training
            text = text + tokenizer.eos_token
            texts.append(text)
        return {"text": texts}

    dataset = dataset.map(format_prompts, batched=True)
    return dataset

def train_legal_model():
    print("🚀 Setting up Legal AI model training...")

    # Setup model and tokenizer
    model, tokenizer = setup_model_for_training()

    # Prepare dataset
    dataset = prepare_legal_dataset()

    # Training arguments optimized for RTX 3060 Ti
    training_args = TrainingArguments(
        per_device_train_batch_size=1,       # Small batch for 8GB VRAM
        gradient_accumulation_steps=4,        # Effective batch size = 4
        warmup_steps=5,
        max_steps=100,                        # Adjust based on dataset size
        learning_rate=2e-4,
        fp16=not torch.cuda.is_bf16_supported(),
        bf16=torch.cuda.is_bf16_supported(),
        logging_steps=1,
        optim="adamw_8bit",                   # Memory efficient optimizer
        weight_decay=0.01,
        lr_scheduler_type="linear",
        seed=3407,
        output_dir="./legal_ai_model_output",
        save_strategy="steps",
        save_steps=50,
        save_total_limit=2,
    )

    # SFT Trainer
    trainer = SFTTrainer(
        model=model,
        tokenizer=tokenizer,
        train_dataset=dataset["train"],
        dataset_text_field="text",
        max_seq_length=2048,
        dataset_num_proc=2,
        packing=False,
        args=training_args,
    )

    # Start training
    print("🔥 Starting training...")
    trainer_stats = trainer.train()

    print("✅ Training completed!")
    print(f"Training loss: {trainer_stats.training_loss:.4f}")

    # Save the model
    model.save_pretrained("legal_ai_lora_model")
    tokenizer.save_pretrained("legal_ai_lora_model")

    return model, tokenizer

if __name__ == "__main__":
    model, tokenizer = train_legal_model()
```

from unsloth import FastLanguageModel
import torch

def merge_and_save_final_model():
    # Load trained model
    model, tokenizer = FastLanguageModel.from_pretrained(
        model_name="legal_ai_lora_model",
        max_seq_length=2048,
        dtype=None,
        load_in_4bit=True,
    )

    # Merge LoRA weights into base model
    FastLanguageModel.for_inference(model)  # Enable inference mode

    # Save merged model
    model.save_pretrained_merged(
        "legal_ai_merged_model",
        tokenizer,
        save_method="merged_16bit",  # or "merged_4bit" for smaller size
    )

    # Check final model size
    import os
    model_size = sum(os.path.getsize(os.path.join("legal_ai_merged_model", f))
                    for f in os.listdir("legal_ai_merged_model"))
    print(f"Final model size: {model_size / 1024 / 1024:.1f} MB")

if __name__ == "__main__":
    merge_and_save_final_model()

4.1 Fast Inference Script
-------------------------
```python
# legal_ai_inference.py
from unsloth import FastLanguageModel
import torch

class LegalAIInference:
    def __init__(self, model_path="legal_ai_merged_model"):
        self.model, self.tokenizer = FastLanguageModel.from_pretrained(
            model_name=model_path,
            max_seq_length=2048,
            dtype=None,
            load_in_4bit=True,
        )
        FastLanguageModel.for_inference(self.model)  # Enable fast inference

    def analyze_contract(self, contract_text):
        prompt = f"""### Instruction:
Analyze this contract clause for legal risks and provide a detailed risk assessment.

### Input:
{contract_text}

### Response:
"""

        inputs = self.tokenizer(prompt, return_tensors="pt")

        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=256,
                temperature=0.7,
                do_sample=True,
                use_cache=True,
                pad_token_id=self.tokenizer.eos_token_id
            )

        response = self.tokenizer.decode(outputs[0][inputs['input_ids'].shape[1]:],
                                       skip_special_tokens=True)
        return response

# Usage example
if __name__ == "__main__":
    legal_ai = LegalAIInference()

    test_contract = """
    The contractor shall deliver all work product no later than
    December 31, 2024, time being of the essence. Any delay
    shall result in immediate termination.
    """

    analysis = legal_ai.analyze_contract(test_contract)
    print("Legal Analysis:")
    print(analysis)
```

# validate_model.py
def validate_legal_model():
    legal_ai = LegalAIInference()

    test_cases = [
        {
            "input": "Time is of the essence delivery clause",
            "expected_risk": "high",
            "expected_keywords": ["material breach", "immediate", "deadline"]
        },
        {
            "input": "Standard confidentiality agreement",
            "expected_risk": "low",
            "expected_keywords": ["confidential", "protection", "disclosure"]
        }
    ]

    results = []
    for case in test_cases:
        analysis = legal_ai.analyze_contract(case["input"])

        # Simple validation
        risk_detected = any(keyword in analysis.lower()
                          for keyword in case["expected_keywords"])

        results.append({
            "input": case["input"],
            "analysis": analysis,
            "validation_passed": risk_detected
        })

    accuracy = sum(r["validation_passed"] for r in results) / len(results)
    print(f"Validation accuracy: {accuracy:.2%}")

    return results

if __name__ == "__main__":

STEP 8: OPTIMIZATION TIPS
=========================

8.1 Reducing Model Size Further
-------------------------------
- Use rank=8 instead of rank=16: Saves ~50MB
- Quantize to int8: Additional 2x compression
- Prune attention heads: 10-20% size reduction
- Knowledge distillation: Transfer to smaller model

8.2 Training Optimizations
--------------------------
- Use gradient checkpointing: Reduces VRAM usage
- Enable mixed precision (fp16/bf16): Faster training
- Use DeepSpeed ZeRO: For larger datasets
- Batch size tuning: Find optimal batch size for your GPU

8.3 Inference Optimizations
---------------------------
- Enable torch.compile: 20-30% speedup
- Use ONNX export: Cross-platform deployment
- TensorRT optimization: Additional speedup
- Quantization to int8: Faster inference

TROUBLESHOOTING
==============

Common Issues:
1. CUDA out of memory → Reduce batch size to 1
2. Training too slow → Increase gradient accumulation
3. Model too large → Reduce LoRA rank or use more quantization
4. Poor quality → Increase training steps or improve dataset
