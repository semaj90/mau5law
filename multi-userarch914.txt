## TECHNICAL SPECIFICATIONS

### WebSocket Architecture
```typescript
// Server: src/lib/server/websocket-server.ts
class CollaborationWebSocketServer {
  private redis: Redis
  private wss: WebSocketServer

  constructor() {
    this.wss = new WebSocketServer({
      port: 7173,  // Use existing configured WS ports
      path: '/collaboration'
    })
    this.setupRedisSubscriptions()
  }

  private async setupRedisSubscriptions() {
    await this.redis.psubscribe('case:*:*')
    this.redis.on('pmessage', this.broadcastToClients.bind(this))
  }
}
```
