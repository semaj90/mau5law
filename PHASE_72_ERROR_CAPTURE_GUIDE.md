# Phase 72: Real-Time Error Capture & Brain Integration\n\n**Date:** December 2, 2025  \n**Goal:** Wire CLI errors → Phase 72 Brain → Ollama recommendations → Claude/Copilot/Gemini\n\n---\n\n## 🎯 What We're Building\n\nA system that:\n1. **Captures** CLI errors from `npm run dev:quic` (Vite HMR, TypeScript, Svelte)\n2. **Sends** them to Phase 72 brain (legal_ai_db)\n3. **Gets** AI recommendations from Ollama (gemma3-legal)\n4. **Displays** in NES Command Center or sends to Claude/Copilot/Gemini\n\n---\n\n## 🚀 Step 1: Create Error Capture Service\n\nCreate `sveltekit-frontend/src/lib/server/error-capture.ts`:\n\n```typescript\n// src/lib/server/error-capture.ts\nimport { pool } from './db';\n\nexport interface CapturedError {\n  code: string;\n  message: string;\n  file_path?: string;\n  line?: number;\n  column?: number;\n  severity: 'error' | 'warning' | 'info';\n  source: 'vite' | 'typescript' | 'svelte' | 'cli';\n  timestamp: string;\n}\n\nexport async function captureError(error: CapturedError): Promise<void> {\n  const client = await pool.connect();\n  try {\n    const errorHash = `${error.code}:${error.file_path}:${error.line}`;\n    \n    await client.query(\n      `\n      INSERT INTO phase72_error (\n        error_hash, file_path, line, column, code, severity, message, phase, cycle\n      ) VALUES ($1, $2, $3, $4, $5, $6, $7, 72, 1)\n      ON CONFLICT (error_hash) DO UPDATE SET\n        created_at = now()\n      `,\n      [\n        errorHash,\n        error.file_path || 'unknown',\n        error.line || 0,\n        error.column || 0,\n        error.code,\n        error.severity,\n        error.message,\n      ]\n    );\n  } finally {\n    client.release();\n  }\n}\n\nexport async function captureErrorBatch(errors: CapturedError[]): Promise<void> {\n  for (const error of errors) {\n    await captureError(error);\n  }\n}\n```\n\n---\n\n## 🚀 Step 2: Create CLI Error Parser\n\nCreate `sveltekit-frontend/scripts/parse-cli-errors.mjs`:\n\n```javascript\n// scripts/parse-cli-errors.mjs\nimport fs from 'fs';\nimport readline from 'readline';\nimport fetch from 'node-fetch';\n\nconst API_URL = 'http://localhost:5173/api/phase72/capture-error';\n\nconst errorPatterns = [\n  // Vite HMR errors\n  {\n    pattern: /\\[vite\\]\\s+(.+?)\\s+at\\s+(.+?):(\\d+):(\\d+)/,\n    parse: (match) => ({\n      code: 'VITE_HMR',\n      message: match[1],\n      file_path: match[2],\n      line: parseInt(match[3]),\n      column: parseInt(match[4]),\n      source: 'vite'\n    })\n  },\n  // TypeScript errors\n  {\n    pattern: /error TS(\\d+):\\s+(.+?)\\s+at\\s+(.+?):(\\d+):(\\d+)/,\n    parse: (match) => ({\n      code: `TS${match[1]}`,\n      message: match[2],\n      file_path: match[3],\n      line: parseInt(match[4]),\n      column: parseInt(match[5]),\n      source: 'typescript'\n    })\n  },\n  // Svelte errors\n  {\n    pattern: /\\[svelte\\]\\s+(.+?)\\s+in\\s+(.+?):(\\d+):(\\d+)/,\n    parse: (match) => ({\n      code: 'SVELTE_ERROR',\n      message: match[1],\n      file_path: match[2],\n      line: parseInt(match[3]),\n      column: parseInt(match[4]),\n      source: 'svelte'\n    })\n  }\n];\n\nasync function parseAndSendError(line) {\n  for (const { pattern, parse } of errorPatterns) {\n    const match = line.match(pattern);\n    if (match) {\n      const error = parse(match);\n      error.severity = 'error';\n      error.timestamp = new Date().toISOString();\n      \n      try {\n        await fetch(API_URL, {\n          method: 'POST',\n          headers: { 'Content-Type': 'application/json' },\n          body: JSON.stringify(error)\n        });\n        console.log(`[phase72] Captured: ${error.code} in ${error.file_path}`);\n      } catch (err) {\n        console.error(`[phase72] Failed to send error: ${err.message}`);\n      }\n      break;\n    }\n  }\n}\n\n// Read from stdin\nconst rl = readline.createInterface({\n  input: process.stdin,\n  output: process.stdout,\n  terminal: false\n});\n\nrl.on('line', (line) => {\n  parseAndSendError(line);\n});\n\nconsole.log('[phase72] Error capture service started. Listening for CLI errors...');\n```\n\n---\n\n## 🚀 Step 3: Create Capture API Route\n\nCreate `sveltekit-frontend/src/routes/api/phase72/capture-error/+server.ts`:\n\n```typescript\n// src/routes/api/phase72/capture-error/+server.ts\nimport { json } from '@sveltejs/kit';\nimport type { RequestHandler } from './$types';\nimport { pool } from '$lib/server/db';\n\ntype CaptureErrorBody = {\n  code: string;\n  message: string;\n  file_path?: string;\n  line?: number;\n  column?: number;\n  severity: 'error' | 'warning' | 'info';\n  source: 'vite' | 'typescript' | 'svelte' | 'cli';\n  timestamp?: string;\n};\n\nexport const POST: RequestHandler = async ({ request }) => {\n  const body = (await request.json()) as CaptureErrorBody;\n\n  const client = await pool.connect();\n  try {\n    const errorHash = `${body.code}:${body.file_path}:${body.line}`;\n\n    await client.query(\n      `\n      INSERT INTO phase72_error (\n        error_hash, file_path, line, column, code, severity, message, phase, cycle\n      ) VALUES ($1, $2, $3, $4, $5, $6, $7, 72, 1)\n      ON CONFLICT (error_hash) DO UPDATE SET\n        created_at = now()\n      `,\n      [\n        errorHash,\n        body.file_path || 'unknown',\n        body.line || 0,\n        body.column || 0,\n        body.code,\n        body.severity,\n        body.message\n      ]\n    );\n\n    return json({ success: true, error_hash: errorHash });\n  } catch (err) {\n    console.error('Error capturing error:', err);\n    return json(\n      { success: false, error: 'Failed to capture error' },\n      { status: 500 }\n    );\n  } finally {\n    client.release();\n  }\n};\n```\n\n---\n\n## 🚀 Step 4: Wire to npm run dev:quic\n\nUpdate `package.json`:\n\n```json\n{\n  \"scripts\": {\n    \"dev:quic\": \"npm run dev 2>&1 Already captured via API route\n  console.log(`[phase72] Error captured: ${error.code}`);\n}\n\nasync function sendToClaude(error, apiKey) {\n  const response = await fetch(CLAUDE_API, {\n    method: 'POST',\n    headers: {\n      'x-api-key': apiKey,\n      'anthropic-version': '2023-06-01',\n      'content-type': 'application/json'\n    },\n    body: JSON.stringify({\n      model: ${error.file_path}:${error.line}:${error.column}`\n        }\n      ]\n    })\n  });\n\n  const data = await response.json();\n  console.log('[claude]', data.content[0].text);\n}\n\nasync function }\n          ]\n        }\n      ]\n    })\n  });\n\n  const data = await response.json();\n  console.log('[gemini]', data.candidates[0].content.parts[0].text);\n}\n\nexport { sendToPhase72Brain,
