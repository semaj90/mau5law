import ollama
import json
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

def export_gemma3_270m():
    """Extract gemma3:270m from Ollama for training"""

    # Get model info from Ollama
    model_info = ollama.show('gemma3:270m')
    print(f"Model info: {model_info}")

    # Ollama models are stored in:
    # Windows: C:\Users\{user}\.ollama\models\blobs\
    # We need to convert to HuggingFace format

    # Alternative: Use the official Gemma3 model as base
    model_name = "google/gemma-2-270m"  # Official HF version

    try:
        tokenizer = AutoTokenizer.from_pretrained(model_name)
        model = AutoModelForCausalLM.from_pretrained(
            model_name,
            torch_dtype=torch.float16,
            device_map="auto"
        )

        print("✅ Model loaded successfully")
        print(f"Model size: {sum(p.numel() for p in model.parameters())} parameters")

        return model, tokenizer

    except Exception as e:
        print(f"Error loading model: {e}")
        print("Using Ollama API instead...")
        return None, None

if __name__ == "__main__":
    model, tokenizer = export_gemma3_270m()

2.1 Training Script Using Existing Models

```python
# train_with_ollama_models.py
import torch
from unsloth import FastLanguageModel
from transformers import TrainingArguments
from trl import SFTTrainer
from datasets import load_dataset
import ollama

class OllamaLegalTrainer:
    def __init__(self):
        # Use gemma3:270m as base (closest to 500MB target)
        self.base_model = "google/gemma-2-270m"  # Official HF version
        self.legal_model = "gemma3-legal:latest"  # Your expert model

    def setup_training_model(self):
        """Setup 270m model for training"""
        model, tokenizer = FastLanguageModel.from_pretrained(
            model_name=self.base_model,
            max_seq_length=2048,
            dtype=None,
            load_in_4bit=True,  # Quantize for memory efficiency
        )

        # Add LoRA for efficient fine-tuning
        model = FastLanguageModel.get_peft_model(
            model,
            r=8,  # Lower rank for 270m model
            target_modules=["q_proj", "k_proj", "v_proj", "o_proj"],
            lora_alpha=16,
            lora_dropout=0.1,
            bias="none",
            use_gradient_checkpointing="unsloth",
            random_state=3407,
        )

        return model, tokenizer

    def generate_training_data_with_ollama(self, num_samples=100):
        """Generate training data using your gemma3-legal model"""

        legal_scenarios = [
            "contract termination clauses",
            "liability limitation provisions",
            "confidentiality agreements",
            "payment terms and penalties",
            "intellectual property rights",
            "force majeure clauses",
            "dispute resolution mechanisms",
            "warranty and representation terms"
        ]

        training_data = []

        for i in range(num_samples):
            scenario = legal_scenarios[i % len(legal_scenarios)]

            # Generate with your expert legal model
            prompt = f"Provide a detailed legal analysis of {scenario} including risks and recommendations"

            response = ollama.generate(
                model=self.legal_model,
                prompt=prompt,
                options={'temperature': 0.7, 'num_predict': 300}
            )

            # Format for training
            formatted_text = f"""### Instruction:
Analyze the legal implications and provide recommendations.

### Input:
{scenario}

### Response:
{response['response']}"""

            training_data.append({"text": formatted_text})

            if (i + 1) % 10 == 0:
                print(f"Generated {i + 1}/{num_samples} samples")

        return training_data

    def train_model(self):
        """Train the 270m model with legal data"""
        print("🚀 Setting up training...")

        # Setup model
        model, tokenizer = self.setup_training_model()

        # Generate training data from your expert model
        print("📚 Generating training data from gemma3-legal...")
        training_data = self.generate_training_data_with_ollama(200)

        # Save dataset
        import json
        with open("ollama_generated_legal_data.jsonl", "w") as f:
            for item in training_data:
                f.write(json.dumps(item) + "\n")

        # Load as dataset
        dataset = load_dataset("json", data_files="ollama_generated_legal_data.jsonl")

        # Training arguments for RTX 3060 Ti
        training_args = TrainingArguments(
            per_device_train_batch_size=2,      # 270m model allows larger batch
            gradient_accumulation_steps=2,      # Effective batch = 4
            warmup_steps=10,
            max_steps=100,                      # Quick training
            learning_rate=2e-4,
            fp16=True,
            logging_steps=5,
            optim="adamw_8bit",
            weight_decay=0.01,
            lr_scheduler_type="linear",
            seed=3407,
            output_dir="./legal_270m_output",
            save_strategy="steps",
            save_steps=25,
        )

        # Trainer
        trainer = SFTTrainer(
            model=model,
            tokenizer=tokenizer,
            train_dataset=dataset["train"],
            dataset_text_field="text",
            max_seq_length=2048,
            args=training_args,
        )

        # Train
        print("🔥 Starting training...")
        trainer.train()

        # Save
        model.save_pretrained("legal_270m_finetuned")
        tokenizer.save_pretrained("legal_270m_finetuned")

        print("✅ Training complete!")
        return model, tokenizer

if __name__ == "__main__":
    trainer = OllamaLegalTrainer()
    model, tokenizer = trainer.train_model()

# embedding_integration.py
import ollama
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

class LegalEmbeddingEngine:
    def __init__(self):
        # Use your existing embedding models
        self.primary_embedder = "embeddinggemma:latest"    # 621MB
        self.backup_embedder = "nomic-embed-text:latest"   # 274MB

    def get_legal_embeddings(self, text, use_backup=False):
        """Generate embeddings using your Ollama models"""

        model = self.backup_embedder if use_backup else self.primary_embedder

        try:
            response = ollama.embeddings(model=model, prompt=text)
            return np.array(response['embedding'])
        except Exception as e:
            print(f"Error with {model}, trying backup...")
            if not use_backup:
                return self.get_legal_embeddings(text, use_backup=True)
            else:
                raise e

    def find_similar_contracts(self, query_text, contract_database):
        """Find similar contracts using your embeddings"""

        query_embedding = self.get_legal_embeddings(query_text)

        similarities = []
        for contract in contract_database:
            contract_embedding = self.get_legal_embeddings(contract['text'])
            similarity = cosine_similarity(
                query_embedding.reshape(1, -1),
                contract_embedding.reshape(1, -1)
            )[0][0]
            similarities.append((contract, similarity))

        # Return top 5 most similar
        similarities.sort(key=lambda x: x[1], reverse=True)
        return similarities[:5]

# Usage
embedder = LegalEmbeddingEngine()
similar_contracts = embedder.find_similar_contracts(
    "termination clause analysis",
    contract_database
)

# ollama_legal_inference.py
import ollama
import time

class OllamaLegalAI:
    def __init__(self):
        self.models = {
            'expert': 'gemma3-legal:latest',    # 7.3GB - Complex analysis
            'fast': 'gemma3:270m',              # 291MB - Quick checks
            'embeddings': 'embeddinggemma:latest'
        }

    def analyze_contract(self, contract_text, complexity='auto'):
        """Analyze contract with appropriate model"""

        # Auto-detect complexity
        if complexity == 'auto':
            word_count = len(contract_text.split())
            complexity = 'complex' if word_count > 500 else 'simple'

        # Choose model based on complexity
        model = self.models['expert'] if complexity == 'complex' else self.models['fast']

        prompt = f"""Analyze this contract for legal risks:

Contract: {contract_text}

Provide:
1. Risk assessment (1-10 scale)
2. Key concerns
3. Recommendations

Analysis:"""

        start_time = time.time()

        response = ollama.generate(
            model=model,
            prompt=prompt,
            options={
                'temperature': 0.3,  # More deterministic for legal analysis
                'top_p': 0.9,
                'num_predict': 300
            }
        )

        inference_time = time.time() - start_time

        return {
            'analysis': response['response'],
            'model_used': model,
            'inference_time': inference_time,
            'complexity': complexity
        }

    def quick_risk_check(self, clause_text):
        """Fast risk assessment using 270m model"""

        prompt = f"Risk level (1-10) for: {clause_text}\nRisk:"

        response = ollama.generate(
            model=self.models['fast'],
            prompt=prompt,
            options={'temperature': 0.1, 'num_predict': 50}
        )

        return response['response']

    def get_embeddings(self, text):
        """Get embeddings using your existing model"""

        response = ollama.embeddings(
            model=self.models['embeddings'],
            prompt=text
        )

        return response['embedding']

# Usage example
legal_ai = OllamaLegalAI()

test_contract = """
The contractor shall deliver all work product no later than
December 31, 2024, time being of the essence. Any delay
shall result in immediate termination.
"""

# Analyze with auto model selection
result = legal_ai.analyze_contract(test_contract)
print(f"Analysis ({result['model_used']}): {result['analysis']}")
print(f"Time: {result['inference_time']:.2f}s")

# Quick risk check
risk = legal_ai.quick_risk_check("time being of essence clause")
print(f"Quick risk: {risk}")
```

# distill_legal_knowledge.py
import ollama

def distill_legal_knowledge():
    """Distill knowledge from gemma3-legal to gemma3:270m"""

    # Test scenarios
    legal_scenarios = [
        "liability limitation clauses",
        "termination provisions",
        "confidentiality agreements",
        "payment terms",
        "force majeure clauses"
    ]

    distillation_data = []

    for scenario in legal_scenarios:
        # Get expert response from large model
        expert_response = ollama.generate(
            model='gemma3-legal:latest',
            prompt=f"Provide detailed legal analysis of {scenario}",
            options={'temperature': 0.3, 'num_predict': 400}
        )

        # Create training pair for small model
        training_pair = {
            "input": f"Analyze {scenario}",
            "target": expert_response['response']
        }

        distillation_data.append(training_pair)
