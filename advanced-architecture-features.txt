async sendMessage(caseId: string, content: string, options?: SendOptions): Promise<AIMessage> {

  try {
    // Pre-call optimization: Select optimal backend based on metrics
    const backend = await this.selectOptimalBackend(content, options?.legalContext);

    // Performance-aware context building
    const contextMessages = await this.buildSmartContext(caseId, content);

    // Server call with acceleration decision
    const response = options?.useAcceleration
      ? await this.sendWithAcceleration(content, contextMessages, backend)
      : await this.sendToBackend(backend, contextMessages);

    // Post-call metrics update
    this.updateMetrics(backend, performance.now() - startTime);

    return assistantMessage;
  } catch (error) {
    // Performance-based fallback selection
    if (this.config.autoSwitchBackend) {
      const fallbackBackends = this.availableBackends
        .filter(b => b !== this.currentBackend)
        .sort((a, b) => this.metrics.backendLatency[a] - this.metrics.backendLatency[b]);

      // Try fastest backends first
      for (const fallbackBackend of fallbackBackends) {
        try {
          return await this.sendMessage(caseId, content, { ...options, backend: fallbackBackend });
        } catch (fallbackError) {
          console.error(`❌ Fallback ${fallbackBackend} failed:`, fallbackError);
        }
      }
    }
  }
}

### Intelligent Backend Selection
```typescript
private async selectOptimalBackend(message: string, context?: string): Promise<Backend> {
  const complexity = this.analyzeMessageComplexity(message);
  const hasLegalContext = this.hasLegalContext(message, context);
  const requiresSpeed = this.requiresSpeedOptimization(message);

  // Multi-factor scoring algorithm
  const scores: Record<Backend, number> = {
    'vllm': this.calculateBackendScore('vllm', complexity, hasLegalContext, requiresSpeed),
    'ollama': this.calculateBackendScore('ollama', complexity, hasLegalContext, requiresSpeed),
    'webasm': this.calculateBackendScore('webasm', complexity, hasLegalContext, requiresSpeed),
    'go-micro': this.calculateBackendScore('go-micro', complexity, hasLegalContext, requiresSpeed)
  };

  // Performance-based selection considering:
  // - Current backend health (40% weight)
  // - Complexity optimization (30% weight)
  // - Legal context bonus (20% weight)
  // - Speed requirements (20% weight)
  // - Latency penalty (up to -30%)
}

private calculateBackendScore(backend: Backend, complexity: string, hasLegalContext: boolean, requiresSpeed: boolean): number {
  let score = this.backendHealth[backend] * 0.4; // Base health

  // Complexity preferences - each backend optimized for different loads
  const complexityScores = {
    'vllm': { simple: 0.7, medium: 0.9, complex: 1.0 },      // Best for complex queries
    'ollama': { simple: 0.9, medium: 0.8, complex: 0.9 },    // Balanced performance
    'webasm': { simple: 1.0, medium: 0.6, complex: 0.3 },    // Fast simple queries
    'go-micro': { simple: 0.6, medium: 0.8, complex: 1.0 }   // Scalable architecture
  };
  score += complexityScores[backend][complexity] * 0.3;

  // Legal context bonuses
  if (hasLegalContext) {
    const legalBonuses = { 'vllm': 0.2, 'ollama': 0.3, 'webasm': 0.1, 'go-micro': 0.3 };
    score += legalBonuses[backend];
  }

  // Speed optimization
  if (requiresSpeed) {
    const speedScores = { 'vllm': 0.6, 'ollama': 0.8, 'webasm': 1.0, 'go-micro': 0.7 };
    score += speedScores[backend] * 0.2;
  }

  // Real-time latency penalty
  const latencyPenalty = Math.min(this.metrics.backendLatency[backend] / 5000, 0.3);
  score -= latencyPenalty;

  return Math.max(0, Math.min(1, score));
}
```

private async sendWithAcceleration(content: string, contextMessages: AIMessage[], backend: Backend) {
  const complexity = this.analyzeMessageComplexity(content);

  // Smart acceleration routing
  const useLocalAI = complexity === 'simple' && content.length < 200;
  const useCUDAService = complexity === 'complex' || backend === 'go-micro';

  if (useLocalAI) {
    // Browser-local Gemma 270M for simple queries
    return await this.sendWithLocalBrowserAI(content, contextMessages);
  } else if (useCUDAService) {
    // CUDA TensorRT for complex legal analysis
    return await this.sendWithCUDAService(content, contextMessages);
  } else {
    // SIMD + WebGPU hybrid acceleration
    return await this.sendWithSIMDWebGPU(content, contextMessages);
  }
}

## 🧠 Context & Memory Management

### Smart Context Building with Caching
```typescript
private async buildSmartContext(caseId: string, query: string, legalContext?: string): Promise<AIMessage[]> {
  const cacheKey = `${caseId}-${query}-${legalContext || ''}`;

  // Performance: Return cached context if available
  if (this.contextCache.has(cacheKey)) {
    return this.contextCache.get(cacheKey)!;
  }

  const caseMessages = this.cases[caseId]?.messages || [];

  // Intelligent context selection:
  // 1. Recent messages (last 10) for temporal relevance
  const recentMessages = caseMessages.slice(-10);

  // 2. Semantic relevance based on query content
  const relevantMessages = caseMessages.filter(msg => {
    const queryLower = query.toLowerCase();
    return msg.content.toLowerCase().includes(queryLower) ||
           (legalContext && msg.content.toLowerCase().includes(legalContext.toLowerCase()));
  }).slice(0, 5);

  // 3. Deduplicate and sort chronologically
  const contextMessages = [...new Map(
    [...recentMessages, ...relevantMessages].map(msg => [msg.id, msg])
  ).values()].sort((a, b) => a.timestamp - b.timestamp);

  // Cache management with automatic cleanup
  this.contextCache.set(cacheKey, contextMessages);
  if (this.contextCache.size > 100) {
    const keys = Array.from(this.contextCache.keys());
    keys.slice(0, 50).forEach(key => this.contextCache.delete(key));
  }

  return contextMessages;
}
```

private startHealthMonitoring() {
  setInterval(async () => {
    try {
      const response = await fetch('/api/ai/health');
      const healthData = await response.json();

      // Real-time backend health scores
      this.backendHealth = {
        'vllm': healthData.backends?.vllm?.reachable ? 1.0 : 0.0,
        'ollama': healthData.backends?.ollama?.version ? 1.0 : 0.0,
        'webasm': healthData.backends?.webasm?.loaded ? 1.0 : 0.0,
        'go-micro': healthData.backends?.['go-micro']?.healthy ? 1.0 : 0.0
      };

      // Automatically update available backends list
      this.availableBackends = Object.entries(this.backendHealth)
        .filter(([_, score]) => score > 0.1)
        .map(([backend]) => backend as Backend);

    } catch (error) {
      console.warn('Health check failed:', error);
    }
  }, 30000); // Every 30 seconds
}

private formatBackendPayload(backend: Backend, messages: AIMessage[]) {
  const basePayload = {
    messages: messages.map(msg => ({ role: msg.role, content: msg.content })),
    temperature: this.config.temperature,
    model: this.config.model
  };

  // Backend-specific optimizations
  switch (backend) {
    case 'vllm':
      return {
        ...basePayload,
        openaiModel: 'mistralai/Mistral-7B-Instruct-v0.3',
        max_tokens: 2048,
        stop: ['</s>']
      };

    case 'webasm':
      return {
        ...basePayload,
        useWASM: true,
        enableGPU: true,
        quantization: 'int8',
        cache: true
      };

    case 'go-micro':
      return {
        ...basePayload,
        service: 'legal-analysis',
        priority: 'high',
        timeout: 5000,
        retries: 2
      };

    default:
      return basePayload;
  }
}
