#!/usr/bin/env python3
"""
Build INT4 AWQ TensorRT-LLM engine for Gemma3 Legal Model
Direct from FP16 checkpoint (no external AWQ needed).
"""

import subprocess
from pathlib import Path

MODEL_DIR = Path("/home/james/gemma3_checkpoint_working")  # FP16 shards
OUTPUT_DIR = Path("/home/james/gemma3_trtllm_int4")
MAX_BATCH_SIZE = 2
MAX_INPUT_LEN = 1024
MAX_SEQ_LEN = 2048
MAX_BEAM_WIDTH = 1

def build_engine():
    build_cmd = [
        "trtllm-build",
        f"--checkpoint_dir={MODEL_DIR}",
        f"--output_dir={OUTPUT_DIR}",
        "--use_gemm_plugin=auto",
        "--use_gpt_attention_plugin=float16",
        f"--max_batch_size={MAX_BATCH_SIZE}",
        f"--max_input_len={MAX_INPUT_LEN}",
        f"--max_seq_len={MAX_SEQ_LEN}",
        f"--max_beam_width={MAX_BEAM_WIDTH}",
        "--use_weight_only",
        "--weight_only_precision=int4_awq",  # Native INT4 quant
        "--per_group",
        "--group_size=128",
        "--int8_kv_cache",                   # ✅ required for 8 GB VRAM
        "--enable_context_fmha",             # FlashAttention
        "--paged_kv_cache",
        "--remove_input_padding",
        "--tp_size=1",
        "--pp_size=1",
        "--dtype=float16",
    ]

    print("🔧 Running build command:")
    print(" ".join(build_cmd))
    subprocess.run(build_cmd, check=True)

if __name__ == "__main__":
    build_engine()
