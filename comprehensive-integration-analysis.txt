class CyberWarElephantGraphEngine implements WasmGraphEngine {

                embeddings: processedEvidence.embeddings,

                nodes: processedEvidence.entities,

                bvhVisualization: bvhVisualization,
                metadata: {
                    source: 'neo4j-bvh-enhanced',
                    evidenceProcessed: true,

                }
            };
