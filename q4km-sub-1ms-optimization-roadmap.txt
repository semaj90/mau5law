// Persistent GPU memory allocation
static float* persistent_input_buffer = nullptr;
static float* persistent_output_buffer = nullptr;
static float* persistent_workspace = nullptr;

void initializePersistentMemory() {
    // Allocate once at startup, never free
    cudaMalloc(&persistent_input_buffer, MAX_INPUT_SIZE);
    cudaMalloc(&persistent_output_buffer, MAX_OUTPUT_SIZE);
    cudaMalloc(&persistent_workspace, WORKSPACE_SIZE);

    // Keep GPU memory warm
    cudaMemset(persistent_input_buffer, 0, MAX_INPUT_SIZE);
    cudaMemset(persistent_output_buffer, 0, MAX_OUTPUT_SIZE);
}

// Ultra-fast inference with zero memory allocation
float inferWithPersistentMemory(const float* input, float* output) {
    // No cudaMalloc/cudaFree - use persistent buffers
    cudaMemcpyAsync(persistent_input_buffer, input, input_size,
                    cudaMemcpyHostToDevice, stream);

    // Execute with persistent memory
    executeKernel(persistent_input_buffer, persistent_output_buffer);

    // Copy result back
    cudaMemcpyAsync(output, persistent_output_buffer, output_size,
                    cudaMemcpyDeviceToHost, stream);

    return measureExecutionTime();
}
```

// Triple-buffered pipeline for continuous processing
struct TripleBuffer {
    float* buffer_A;  // Currently processing
    float* buffer_B;  // Loading next input
    float* buffer_C;  // Preparing output
};

void pipelinedInference() {
    // Stream 1: Data loading
    cudaMemcpyAsync(buffer_B, next_input, size, cudaMemcpyHostToDevice, stream1);

    // Stream 2: Computation (parallel with data loading)
    q4km_kernel<<<grid, block, 0, stream2>>>(buffer_A, output_buffer_A);

    // Stream 3: Result transfer (parallel with computation)
    cudaMemcpyAsync(result, output_buffer_C, size, cudaMemcpyDeviceToHost, stream3);

    // Rotate buffers for next iteration
    std::rotate(buffers.begin(), buffers.begin() + 1, buffers.end());
}
```
